  11: const mmkv = new MMKV();
  12: const LAST_INDEX_KEY = "lastViewedWrapIndex";
  13: 
  14: type ApiWrap = {
  15:   wrap_id: number;
  16:   title_en?: string;
  17:   title_te?: string;
  18:   media_en?: { type: string; variants?: Record<string, string>; thumbnail?: string };
  19:   media_te?: { type: string; variants?: Record<string, string>; thumbnail?: string };
  20:   published_at?: string;
  21:   stats?: any;
  22: };

  24: const measureApproxMbps = async (): Promise<number | null> => {
  25:   try {
  26:     const state = await NetInfo.fetch();
  27:     // RN NetInfo may expose downlink in Mbps on some platforms; if present, use it.
  28:     const dl: any = (state as any)?.details?.downlink;
  29:     if (typeof dl === "number" && dl > 0) return dl;
  30:   } catch {}
  31:   return null; // unknown, fall back to default 720p
  32: };
  33: 
  34: const pickVariantUrl = (variants?: Record<string, string>, mbps?: number | null): string | null => {
  35:   if (!variants) return null;
  36:   let tier: "480p" | "720p" | "1080p" = "720p"; // default
  37:   if (typeof mbps === "number") {
  38:     if (mbps < 1.5) tier = "480p";
  39:     else if (mbps <= 3) tier = "720p";
  40:     else tier = "1080p";
  41:   }
  42:   const chosen = variants[tier] || variants["720p"] || variants["480p"] || variants["1080p"];
  43:   if (__DEV__) console.log(`🟣 Wrap variant selected: ${tier} → ${chosen}`);
  44:   return chosen || null;
  45: };

  47: export default function DailyWrapsScreen() {
  48:   const navigation = useNavigation<any>();
  49:   const { Colors } = useTheme();

  54:   const [refreshing, setRefreshing] = useState(false);

  56:   const [initialIndex, setInitialIndex] = useState<number>(0);
  57: 
  58:   // Handle Android hardware back to go Home

  60:     const sub = BackHandler.addEventListener("hardwareBackPress", () => {
  61:       navigation.navigate("HomeScreen" as any);

  63:     });
  64:     return () => sub.remove();
  65:   }, [navigation]);
  66: 
  67:   // Restore last index
  68:   useEffect(() => {
  69:     try {
  70:       const idx = mmkv.getNumber(LAST_INDEX_KEY);
  71:       if (typeof idx === "number" && idx >= 0) setInitialIndex(idx);
  72:     } catch {}
  73:   }, []);
  74: 
  75:   const fetchWraps = useCallback(async () => {
  76:     try {
  77:       if (!refreshing) setLoading(true);

  80:

  83:           .map((it) => {

  85:             if (!media || media.type !== "video") return null; // skip non-video
  86:             const uri = pickVariantUrl(media.variants, mbps);
  87:             if (!uri) return null;
  88:             return {
  89:               video_id: it.wrap_id,
  90:               _id: it.wrap_id,
  91:               uri,
  92:               title_en: it.title_en,
  93:               title_te: it.title_te,
  94:               title: lang === "te" ? it.title_te || it.title_en : it.title_en || it.title_te,
  95:               thumbnail: media.thumbnail || null,
  96:               published_at: it.published_at,
  97:               stats: it.stats,
  98:             } as WrapItem;
  99:           })
 100:           .filter(Boolean) as WrapItem[];
 101: 
 102:         setItems(mapped);
 103:       } else {
 104:         setItems([]);
 105:       }
 106:     } catch (err) {
 107:       console.error("❌ Error fetching wraps:", err);
 108:       setItems([]);
 109:     } finally {
 110:       setLoading(false);
 111:       setRefreshing(false);
 112:     }
 113:   }, [lang, refreshing]);
 114: 
 115:   useEffect(() => {
 116:     fetchWraps();
 117:   }, [fetchWraps]);
 118: 
 119:   const handleRefresh = useCallback(() => {
 120:     setRefreshing(true);
 121:     fetchWraps();
 122:   }, [fetchWraps]);
 123: 
 124:   const handleIndexChange = useCallback((idx: number) => {
 125:     try {
 126:       mmkv.set(LAST_INDEX_KEY, idx);
 127:     } catch {}
 128:   }, []);
 129: 
 130:   if (loading) {
 131:     return (
 132:       <View style={styles.loader}>
 133:         <ActivityIndicator size="large" color={Colors.lavenderPurple} />
 134:       </View>
 135:     );
 136:   }
 137: 
 138:   return (
 139:     <View style={{ flex: 1, backgroundColor: "#000" }}>
 140:       <DailyWrapsReels
 141:         videos={items}
 142:         initialIndex={initialIndex}
 143:         onIndexChange={handleIndexChange}
 144:         refreshing={refreshing}
 145:         onRefresh={handleRefresh}
 146:       />
 147:     </View>
 148:   );
 149: }
